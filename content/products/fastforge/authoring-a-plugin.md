# Authoring a Plugin

Build a working third-party generator in ~30 minutes. We'll create `fastforge-stripe` as a worked example.

## 1. Project layout

```
fastforge-stripe/
├── pyproject.toml
├── README.md
├── fastforge_stripe/
│   ├── __init__.py
│   └── generator.py
└── tests/
    └── test_generator.py
```

## 2. `pyproject.toml`

The critical piece is the entry-point declaration:

```toml
[project]
name = "fastforge-stripe"
version = "0.1.0"
description = "Stripe billing scaffolding for FastForge"
requires-python = ">=3.10"
dependencies = [
    "fastforge-cli>=0.1.0",
]

[project.entry-points."fastforge.generators"]
stripe = "fastforge_stripe.generator:StripeGenerator"

[build-system]
requires = ["setuptools>=68"]
build-backend = "setuptools.build_meta"
```

Key rules:

- The **left-hand side** (`stripe`) becomes `fastforge add stripe`
- The **right-hand side** points to a class (preferred) or instance
- Depend on `fastforge-cli` so users get it on install

## 3. Implement the generator

```python
# fastforge_stripe/generator.py
from pathlib import Path
from typing import Any

from fastforge.generator_protocol import BaseGenerator


class StripeGenerator(BaseGenerator):
    name = "stripe"
    version = "0.1.0"
    description = "Stripe billing and webhook scaffolding"
    capability_key = "stripe"
    delegatable = True

    def emit_inline(self, project_dir: Path, args: dict[str, Any]) -> dict[str, Any]:
        created: list[str] = []
        modified: list[str] = []

        # 1. Write a Stripe client wrapper
        client_path = project_dir / "app" / "integrations" / "stripe_client.py"
        client_path.parent.mkdir(parents=True, exist_ok=True)
        if not client_path.exists():
            client_path.write_text(self._render_client())
            created.append(str(client_path.relative_to(project_dir)))

        # 2. Write a webhook route
        webhook_path = project_dir / "app" / "api" / "routes" / "stripe_webhook.py"
        webhook_path.parent.mkdir(parents=True, exist_ok=True)
        if not webhook_path.exists():
            webhook_path.write_text(self._render_webhook())
            created.append(str(webhook_path.relative_to(project_dir)))

        # 3. Append env entries
        env_example = project_dir / ".env.example"
        if env_example.exists():
            content = env_example.read_text()
            if "STRIPE_API_KEY" not in content:
                env_example.write_text(content + "\nSTRIPE_API_KEY=\nSTRIPE_WEBHOOK_SECRET=\n")
                modified.append(".env.example")

        return {"status": "ok", "created": created, "modified": modified}

    def capability_schema(self) -> dict[str, Any]:
        return {
            "type": "object",
            "properties": {
                "stripe": {
                    "type": "object",
                    "properties": {
                        "version": {"type": "string"},
                        "webhook_path": {"type": "string", "default": "/webhooks/stripe"},
                    },
                    "required": ["version"],
                }
            },
        }

    def validate(self, project_dir: Path) -> list[str]:
        issues: list[str] = []
        if not (project_dir / "app" / "integrations" / "stripe_client.py").exists():
            issues.append("stripe_client.py missing — re-run `fastforge add stripe`")
        return issues

    # ── private renderers ────────────────────────────────────────────────

    def _render_client(self) -> str:
        return '''"""Stripe client wrapper — generated by fastforge-stripe."""
import os
import stripe

stripe.api_key = os.environ["STRIPE_API_KEY"]
'''

    def _render_webhook(self) -> str:
        return '''"""Stripe webhook route — generated by fastforge-stripe."""
from fastapi import APIRouter, Request, HTTPException
import os, stripe

router = APIRouter(prefix="/webhooks", tags=["stripe"])


@router.post("/stripe")
async def stripe_webhook(request: Request) -> dict:
    payload = await request.body()
    sig = request.headers.get("stripe-signature", "")
    try:
        event = stripe.Webhook.construct_event(
            payload, sig, os.environ["STRIPE_WEBHOOK_SECRET"]
        )
    except Exception as e:
        raise HTTPException(status_code=400, detail=str(e))
    return {"received": event["type"]}
'''
```

## 4. Test it locally

In your plugin repo:

```bash
python3 -m pip install -e .
```

Then in a FastForge-generated project:

```bash
fastforge plugins ls
# stripe should appear

fastforge add stripe
```

## 5. Required vs optional methods

| Method | Required? | When to implement |
|---|---|---|
| `name`, `version`, `description`, `capability_key` | ✅ Required | Always |
| `emit_inline` | ✅ Required | Always — standalone code generation |
| `capability_schema` | 🟡 Recommended | Enables `.fastforge.json` validation |
| `validate` | 🟡 Recommended | Powers `fastforge doctor` integration |
| `emit_delegated` | ⛔ Optional | Only if you support `kind="app"` with a platform lib |
| `emit_into_lib` | ⛔ Optional | Only if you support `kind="lib"` |
| `upgrade` | ⛔ Optional | When you ship breaking changes |

The `BaseGenerator` class provides safe default stubs for everything except `emit_inline`.

## 6. Conventions

- **Naming**: PyPI package = `fastforge-<area>` (e.g. `fastforge-stripe`, `fastforge-clerk`, `fastforge-langfuse`)
- **Entry-point name** = the verb the user types: `fastforge add stripe` → entry-point name is `stripe`
- **Capability key** = the key written into `.fastforge.json` (usually same as entry-point name)
- **Idempotency**: re-running `fastforge add <name>` must never duplicate writes. Always check if files exist.
- **Fail-open**: never crash the user's CLI. Raise informative errors only after destructive intent is confirmed.

## 7. Test your generator

Minimal test:

```python
# tests/test_generator.py
from pathlib import Path
from fastforge_stripe.generator import StripeGenerator


def test_emit_inline_creates_expected_files(tmp_path: Path):
    # Arrange — fake FastForge project shape
    (tmp_path / "app" / "api" / "routes").mkdir(parents=True)
    (tmp_path / "app" / "integrations").mkdir(parents=True)
    (tmp_path / ".env.example").write_text("EXISTING=1\n")

    # Act
    result = StripeGenerator().emit_inline(tmp_path, {})

    # Assert
    assert result["status"] == "ok"
    assert (tmp_path / "app" / "integrations" / "stripe_client.py").exists()
    assert (tmp_path / "app" / "api" / "routes" / "stripe_webhook.py").exists()
    assert "STRIPE_API_KEY" in (tmp_path / ".env.example").read_text()


def test_emit_inline_is_idempotent(tmp_path: Path):
    (tmp_path / "app" / "api" / "routes").mkdir(parents=True)
    (tmp_path / "app" / "integrations").mkdir(parents=True)
    (tmp_path / ".env.example").write_text("")

    StripeGenerator().emit_inline(tmp_path, {})
    second = StripeGenerator().emit_inline(tmp_path, {})

    assert second["created"] == []  # nothing re-created
```
